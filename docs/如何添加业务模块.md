# 如何添加业务模块

> 这是若茗 PaaS 最重要的开发指南。  
> 本文以"订单管理"为示例，演示如何在底座上从零到一添加一个完整的业务模块。

---

## 概览

在若茗底座上添加业务模块，标准流程分为 6 步：

```
① 创建模块目录结构
② 定义数据模型（Model）
③ 编写数据库迁移
④ 定义请求/响应结构（Schema）
⑤ 实现业务逻辑（Service + Repository）
⑥ 注册路由（Router）
[可选] ⑦ 注册事件、权限、菜单
[可选] ⑧ 添加前端页面
```

---

## 示例：添加"订单管理"模块

### 第一步：创建模块目录结构

在 `backend/app/modules/` 下新建 `order/` 目录：

```
backend/app/modules/order/
├── __init__.py
├── models.py        # 数据模型（SQLAlchemy）
├── schemas.py       # 请求/响应结构（Pydantic）
├── repository.py    # 数据库操作
├── service.py       # 业务逻辑
└── router.py        # API 路由
```

```bash
mkdir -p backend/app/modules/order
touch backend/app/modules/order/{__init__,models,schemas,repository,service,router}.py
```

---

### 第二步：定义数据模型

```python
# backend/app/modules/order/models.py

from sqlalchemy import Column, String, Integer, Numeric, Text, ForeignKey
from sqlalchemy.orm import relationship
from app.core.base_model import BaseModel  # 继承底座基类，自动拥有 id/时间戳/软删除/租户字段


class Order(BaseModel):
    """订单表"""
    __tablename__ = "biz_order"   # 业务表建议加 biz_ 前缀，区别于框架系统表 sys_

    order_no    = Column(String(32), unique=True, nullable=False, comment="订单号")
    title       = Column(String(128), nullable=False, comment="订单标题")
    amount      = Column(Numeric(10, 2), nullable=False, default=0, comment="金额")
    status      = Column(Integer, nullable=False, default=1, comment="状态: 1=待处理 2=处理中 3=完成 4=取消")
    remark      = Column(Text, comment="备注")
    created_by  = Column(Integer, ForeignKey("sys_user.id"), comment="创建人ID")

    # 关联关系（可选）
    creator = relationship("User", foreign_keys=[created_by], lazy="select")
```

> **继承 `BaseModel` 自动获得**：
> - `id`（主键，BigInt 自增）
> - `created_at` / `updated_at`（时间戳）
> - `is_deleted`（软删除标记）
> - `tenant_id`（多租户字段）

### 第四步：定义 Schema（请求/响应结构）

```python
# backend/app/modules/order/schemas.py

from pydantic import BaseModel, Field
from decimal import Decimal
from datetime import datetime
from typing import Optional


class OrderCreate(BaseModel):
    """创建订单 - 请求体"""
    title:  str     = Field(..., min_length=1, max_length=128, description="订单标题")
    amount: Decimal = Field(..., gt=0, description="金额，必须大于 0")
    remark: Optional[str] = Field(None, max_length=500, description="备注")


class OrderUpdate(BaseModel):
    """更新订单 - 请求体（所有字段可选）"""
    title:  Optional[str]     = Field(None, min_length=1, max_length=128)
    amount: Optional[Decimal] = Field(None, gt=0)
    status: Optional[int]     = Field(None, ge=1, le=4)
    remark: Optional[str]     = None


class OrderResponse(BaseModel):
    """订单详情 - 响应体"""
    id:         int
    order_no:   str
    title:      str
    amount:     Decimal
    status:     int
    remark:     Optional[str]
    tenant_id:  int
    created_at: datetime
    updated_at: datetime
    created_by: Optional[int]

    model_config = {"from_attributes": True}  # 支持 ORM 对象直接转换


class OrderListResponse(BaseModel):
    """订单列表响应（含分页）"""
    items: list[OrderResponse]
    total: int
    page:  int
    size:  int
```

### 第五步：Repository（数据库操作）

```python
# backend/app/modules/order/repository.py

from sqlalchemy import select, func
from sqlalchemy.ext.asyncio import AsyncSession
from app.core.tenant_repository import TenantAwareRepository  # ← 关键！继承此基类
from .models import Order


class OrderRepository(TenantAwareRepository[Order]):
    """
    继承 TenantAwareRepository 后：
    - 所有查询自动加 WHERE tenant_id = :current_tenant_id
    - create/update/delete 自动注入 tenant_id
    - 无需手动写任何 tenant_id 过滤条件
    """

    def __init__(self, db: AsyncSession, tenant_id: int):
        super().__init__(db, Order, tenant_id)

    async def get_by_order_no(self, order_no: str) -> Order | None:
        """按订单号查询（已自动加租户过滤）"""
        result = await self.db.execute(
            self._base_query().where(Order.order_no == order_no)
        )
        return result.scalar_one_or_none()

    async def list_with_pagination(
        self,
        page: int = 1,
        size: int = 20,
        status: int | None = None,
        keyword: str | None = None,
    ) -> tuple[list[Order], int]:
        """分页查询（已自动加租户过滤）"""
        query = self._base_query()

        if status is not None:
            query = query.where(Order.status == status)
        if keyword:
            query = query.where(Order.title.ilike(f"%{keyword}%"))

        # 查总数
        count_result = await self.db.execute(
            select(func.count()).select_from(query.subquery())
        )
        total = count_result.scalar()

        # 查数据
        result = await self.db.execute(
            query.offset((page - 1) * size).limit(size)
        )
        return result.scalars().all(), total
```

### 第六步：Service（业务逻辑）

```python
# backend/app/modules/order/service.py

import uuid
from sqlalchemy.ext.asyncio import AsyncSession
from app.core.exceptions import NotFoundError, BusinessError
from .repository import OrderRepository
from .schemas import OrderCreate, OrderUpdate


class OrderService:

    def __init__(self, db: AsyncSession, tenant_id: int, current_user_id: int):
        self.repo = OrderRepository(db, tenant_id)
        self.current_user_id = current_user_id

    async def create(self, data: OrderCreate) -> dict:
        """创建订单"""
        order = await self.repo.create({
            "order_no":   self._gen_order_no(),
            "title":      data.title,
            "amount":     data.amount,
            "remark":     data.remark,
            "status":     1,
            "created_by": self.current_user_id,
        })
        return order

    async def get(self, order_id: int) -> dict:
        """获取订单详情"""
        order = await self.repo.get_by_id(order_id)
        if not order:
            raise NotFoundError("订单不存在")
        return order

    async def update(self, order_id: int, data: OrderUpdate) -> dict:
        """更新订单"""
        order = await self.get(order_id)  # 自动校验存在性和租户归属

        # 业务规则：已取消的订单不能修改
        if order.status == 4:
            raise BusinessError("已取消的订单不能修改")

        updated = await self.repo.update(order_id, data.model_dump(exclude_none=True))
        return updated

    async def cancel(self, order_id: int) -> dict:
        """取消订单"""
        order = await self.get(order_id)
        if order.status == 4:
            raise BusinessError("订单已取消")
        return await self.repo.update(order_id, {"status": 4})

    async def list(self, page: int, size: int, status: int | None, keyword: str | None):
        """分页查询订单"""
        items, total = await self.repo.list_with_pagination(page, size, status, keyword)
        return {"items": items, "total": total, "page": page, "size": size}

    def _gen_order_no(self) -> str:
        """生成订单号"""
        import time
        return f"ORD{int(time.time() * 1000)}{uuid.uuid4().hex[:6].upper()}"
```

### 第七步：Router（API 路由）

```python
# backend/app/modules/order/router.py

from fastapi import APIRouter, Depends, Query
from sqlalchemy.ext.asyncio import AsyncSession
from app.core.database import get_db
from app.core.auth import get_current_user, CurrentUser
from app.core.response import ApiResponse
from app.core.decorators import audit_log, require_permission
from .service import OrderService
from .schemas import OrderCreate, OrderUpdate, OrderResponse, OrderListResponse

router = APIRouter(prefix="/orders", tags=["订单管理"])


def get_service(
    db: AsyncSession = Depends(get_db),
    current_user: CurrentUser = Depends(get_current_user),
) -> OrderService:
    return OrderService(db, current_user.tenant_id, current_user.id)


@router.post("", response_model=ApiResponse[OrderResponse], summary="创建订单")
@require_permission("order.create")       # 权限校验
@audit_log(action="CREATE", resource="order")   # 操作审计（自动记录）
async def create_order(
    data: OrderCreate,
    svc: OrderService = Depends(get_service),
):
    order = await svc.create(data)
    return ApiResponse.success(order)


@router.get("", response_model=ApiResponse[OrderListResponse], summary="订单列表")
@require_permission("order.list.view")
async def list_orders(
    page:    int = Query(1, ge=1),
    size:    int = Query(20, ge=1, le=100),
    status:  int | None = Query(None),
    keyword: str | None = Query(None),
    svc: OrderService = Depends(get_service),
):
    result = await svc.list(page, size, status, keyword)
    return ApiResponse.success(result)


@router.get("/{order_id}", response_model=ApiResponse[OrderResponse], summary="订单详情")
@require_permission("order.detail.view")
async def get_order(
    order_id: int,
    svc: OrderService = Depends(get_service),
):
    order = await svc.get(order_id)
    return ApiResponse.success(order)


@router.put("/{order_id}", response_model=ApiResponse[OrderResponse], summary="更新订单")
@require_permission("order.update")
@audit_log(action="UPDATE", resource="order")
async def update_order(
    order_id: int,
    data: OrderUpdate,
    svc: OrderService = Depends(get_service),
):
    order = await svc.update(order_id, data)
    return ApiResponse.success(order)


@router.post("/{order_id}/cancel", response_model=ApiResponse[OrderResponse], summary="取消订单")
@require_permission("order.cancel")
@audit_log(action="UPDATE", resource="order", description="取消订单")
async def cancel_order(
    order_id: int,
    svc: OrderService = Depends(get_service),
):
    order = await svc.cancel(order_id)
    return ApiResponse.success(order)
```

### 第八步：注册到主应用

在主路由文件中注册新模块的 router：

```python
# backend/app/main.py（或 backend/app/api/router.py）

from app.modules.order.router import router as order_router

# 在 include_router 列表中添加：
app.include_router(order_router, prefix="/api/v1")
```

完成！访问 `http://localhost:8000/api/docs` 即可看到新增的订单相关接口。

## 底座提供了什么，你不需要自己写

| 你需要写 | 底座帮你处理 |
|---------|------------|
| 业务模型字段 | `id`、时间戳、软删除、`tenant_id` 字段 |
| 业务查询条件 | `tenant_id` 自动过滤（不会查到其他租户数据）|
| 业务接口逻辑 | Token 认证、用户信息注入、权限校验 |
| 具体权限定义 | 权限中间件执行、无权限统一报 403 |
| 业务日志描述 | `created_at`/`updated_at` 自动更新、before/after 数据快照 |
| 具体限流配置 | 接口默认限流、AI Token 配额计数 |
| 错误响应格式 | 统一 `ApiResponse` 格式、错误码体系 |
| 数据库连接管理 | Session 生命周期、连接池管理 |

## 进阶：给模块接入 AI 能力

如果你的业务模块需要 AI 能力，若茗提供了直接调用的方式：

```python
# backend/app/modules/order/service.py

from app.ai.agent.executor import AgentRunner

class OrderService:

    async def analyze_order_risk(self, order_id: int) -> str:
        """用 AI 分析订单风险（调用已配置的 Agent）"""
        order = await self.get(order_id)

        runner = AgentRunner(agent_id=settings.ORDER_RISK_AGENT_ID)
        result = await runner.run(
            user_message=f"请分析以下订单的风险：{order.title}，金额：{order.amount}",
            session_id=f"order-risk-{order_id}",
            scene_type="professional",
        )
        return result["output"]
```

## 进阶：发布和监听事件

```python
# 发布事件（订单创建后通知其他模块）
from app.modules.event_bus.service import event_bus

async def create(self, data: OrderCreate):
    order = await self.repo.create(...)

    # 发布事件，其他模块监听处理（解耦）
    await event_bus.publish("order.created", {
        "order_id": order.id,
        "tenant_id": order.tenant_id,
        "amount": float(order.amount),
    })
    return order


# 在其他模块监听事件
@event_bus.on("order.created")
async def on_order_created(payload: dict):
    # 比如：发送通知、更新统计数据等
    await notification_service.send(
        user_id=...,
        message=f"新订单 {payload['order_id']} 已创建",
    )
```

## 检查清单

完成一个业务模块后，检查以下项目：

- [ ] 数据模型继承 `BaseModel`，表名加 `biz_` 前缀
- [ ] Repository 继承 `TenantAwareRepository`（多租户隔离）
- [ ] 所有接口添加 `@require_permission` 权限装饰器
- [ ] 写操作添加 `@audit_log` 审计装饰器
- [ ] 接口响应使用 `ApiResponse.success()` 统一格式
- [ ] 权限标识符遵循 `{模块}.{资源}.{操作}` 规范
- [ ] 已在后台管理界面配置对应的权限项和菜单
- [ ] 已编写基础单元测试（至少覆盖 Service 层）
- [ ] 迁移文件已人工检查确认
