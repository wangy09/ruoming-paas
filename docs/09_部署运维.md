# 09 部署运维

## 1. Docker Compose（开发/测试环境）

```yaml
# deploy/docker-compose.yml
version: '3.9'

services:
  # PostgreSQL
  postgres:
    image: postgres:15-alpine
    environment:
      POSTGRES_DB: ruoming_saas
      POSTGRES_USER: ruoming
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
    volumes:
      - postgres_data:/var/lib/postgresql/data
    ports:
      - "5432:5432"
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ruoming"]
      interval: 5s
      retries: 5

  # Redis
  redis:
    image: redis:7-alpine
    command: redis-server --requirepass ${REDIS_PASSWORD}
    volumes:
      - redis_data:/data
    ports:
      - "6379:6379"

  # Milvus（向量数据库）
  milvus:
    image: milvusdb/milvus:v2.4.0
    environment:
      ETCD_ENDPOINTS: etcd:2379
      MINIO_ADDRESS: minio:9000
    depends_on:
      - etcd
      - minio
    ports:
      - "19530:19530"

  etcd:
    image: quay.io/coreos/etcd:v3.5.5
    command: etcd -advertise-client-urls=http://127.0.0.1:2379 -listen-client-urls http://0.0.0.0:2379

  minio:
    image: minio/minio:latest
    command: minio server /data
    environment:
      MINIO_ACCESS_KEY: ${MINIO_ACCESS_KEY}
      MINIO_SECRET_KEY: ${MINIO_SECRET_KEY}

  # FastAPI 后端
  backend:
    build: ../backend
    environment:
      DATABASE_URL: postgresql+asyncpg://ruoming:${POSTGRES_PASSWORD}@postgres:5432/ruoming_saas
      REDIS_URL: redis://:${REDIS_PASSWORD}@redis:6379/0
      JWT_SECRET: ${JWT_SECRET}
      ZHUILU_API_KEY: ${ZHUILU_API_KEY}
      ZHUILU_BASE_URL: ${ZHUILU_BASE_URL}
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_started
    ports:
      - "8000:8000"
    volumes:
      - ../backend:/app
      - log_data:/var/log/ruoming-saas

  # Celery Worker
  celery_worker:
    build: ../backend
    command: celery -A app.tasks worker -l info -c 4
    environment:
      DATABASE_URL: postgresql+asyncpg://ruoming:${POSTGRES_PASSWORD}@postgres:5432/ruoming_saas
      REDIS_URL: redis://:${REDIS_PASSWORD}@redis:6379/0
    depends_on:
      - backend

  # Celery Beat（定时任务）
  celery_beat:
    build: ../backend
    command: celery -A app.tasks beat -l info
    depends_on:
      - backend

  # Flower（Celery 监控）
  flower:
    build: ../backend
    command: celery -A app.tasks flower --port=5555
    ports:
      - "5555:5555"

  # Vue 前端（开发模式）
  frontend:
    image: node:20-alpine
    working_dir: /app
    command: sh -c "npm install && npm run dev -- --host 0.0.0.0"
    volumes:
      - ../frontend:/app
    ports:
      - "5173:5173"

  # Nginx
  nginx:
    image: nginx:alpine
    volumes:
      - ./nginx/nginx.conf:/etc/nginx/nginx.conf:ro
      - ./nginx/conf.d:/etc/nginx/conf.d:ro
    ports:
      - "80:80"
      - "443:443"
    depends_on:
      - backend

volumes:
  postgres_data:
  redis_data:
  log_data:
```

## 2. Nginx 配置

```nginx
# deploy/nginx/conf.d/ruoming.conf
server {
    listen 80;
    server_name _;

    # 前端静态资源
    location / {
        root /usr/share/nginx/html;
        try_files $uri $uri/ /index.html;
        expires 7d;
        add_header Cache-Control "public, immutable";
    }

    # API 代理
    location /api/ {
        proxy_pass http://backend:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_read_timeout 300s;  # AI 长请求
    }

    # SSE 流式代理（关闭缓冲）
    location /api/v1/ai/chat {
        proxy_pass http://backend:8000;
        proxy_set_header Connection '';
        proxy_http_version 1.1;
        proxy_buffering off;
        proxy_cache off;
        proxy_read_timeout 300s;
        chunked_transfer_encoding on;
    }

    # WebSocket
    location /ws/ {
        proxy_pass http://backend:8000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
    }
}
```

## 5. 监控告警

### Prometheus 指标

FastAPI 集成 `prometheus-fastapi-instrumentator`，暴露：
- `http_requests_total` - 请求总数
- `http_request_duration_seconds` - 请求耗时
- `http_requests_in_progress` - 进行中请求数

端点：`GET /metrics`

### 告警规则（关键项）

```yaml
# 错误率超过 5% 告警
- alert: HighErrorRate
  expr: rate(http_requests_total{status=~"5.."}[5m]) / rate(http_requests_total[5m]) > 0.05
  for: 2m

# 响应 P99 超过 2s
- alert: SlowAPI
  expr: histogram_quantile(0.99, rate(http_request_duration_seconds_bucket[5m])) > 2

# Celery 任务积压
- alert: CeleryQueueBacklog
  expr: celery_queue_length > 100
```
