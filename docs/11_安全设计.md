## 1. JWT 认证增强

### 1.1 库选型

使用 **PyJWT**（非 `python-jose`）。`python-jose` 已多年未维护，存在已知 CVE，PyJWT 是更活跃的替代方案。

```python
# requirements.txt
PyJWT>=2.8.0       # 替代 python-jose
cryptography>=42.0  # PyJWT 的 RS256/ES256 依赖
```

### 1.2 Token 设计

| Token 类型 | 有效期 | 存储位置 | 用途 |
|-----------|-------|---------|------|
| Access Token | 15 分钟 | 内存（前端 JS 变量） | 接口鉴权 |
| Refresh Token | 7 天 | HTTP-only Cookie 或 安全 localStorage | 刷新 Access Token |

### 1.3 Refresh Token 轮换策略（Rotation）

> **原设计漏洞**：Refresh Token 永久有效，一旦泄露无法吊销。本方案采用轮换策略：每次刷新都签发新的 Refresh Token 并使旧的失效。

```python
# app/modules/auth/service.py
import secrets
import hashlib

class AuthService:

    async def refresh_tokens(self, refresh_token: str, db, redis) -> TokenPair:
        """
        Refresh Token 轮换：
        1. 验证旧 Refresh Token（数据库中查找 hash）
        2. 签发新 Access Token + 新 Refresh Token
        3. 旧 Refresh Token 立即失效（数据库删除 + Redis 黑名单）
        4. 检测重放攻击（旧 Token 已被使用则强制下线所有设备）
        """
        token_hash = hashlib.sha256(refresh_token.encode()).hexdigest()
        stored = await RefreshTokenRepository(db).get_by_hash(token_hash)

        if not stored:
            # 可能是重放攻击：旧 Token 被再次使用
            # 强制吊销该用户所有 Refresh Token
            if payload := self._decode_unsafe(refresh_token):
                await self._revoke_all_tokens(payload.get("sub"), db, redis)
            raise AuthError("Token 无效或已被使用，请重新登录")

        if stored.expires_at < datetime.utcnow():
            await RefreshTokenRepository(db).delete(stored.id)
            raise AuthError("Token 已过期，请重新登录")

        user = await UserRepository(db).get_by_id(stored.user_id)

        # 删除旧 Token
        await RefreshTokenRepository(db).delete(stored.id)

        # 签发新 Token 对
        new_access = self._create_access_token(user)
        new_refresh = secrets.token_urlsafe(64)
        new_refresh_hash = hashlib.sha256(new_refresh.encode()).hexdigest()

        await RefreshTokenRepository(db).create({
            "user_id": user.id,
            "token_hash": new_refresh_hash,
            "expires_at": datetime.utcnow() + timedelta(days=7),
            "user_agent": stored.user_agent,
            "ip": stored.ip,
        })

        return TokenPair(access_token=new_access, refresh_token=new_refresh)

    async def _revoke_all_tokens(self, user_id: int, db, redis):
        """吊销用户所有 Refresh Token（检测到重放攻击时调用）"""
        await RefreshTokenRepository(db).delete_by_user(user_id)
        # 写入 Redis 黑名单，使当前所有 Access Token 失效（JTI 黑名单模式）
        await redis.setex(f"user:force_logout:{user_id}", 900, "1")
```

```sql
-- Refresh Token 存储表
CREATE TABLE sys_refresh_token (
    id          BIGSERIAL PRIMARY KEY,
    user_id     BIGINT NOT NULL,
    token_hash  VARCHAR(64) UNIQUE NOT NULL,    -- SHA256 hash，不存明文
    expires_at  TIMESTAMPTZ NOT NULL,
    user_agent  VARCHAR(256),
    ip          VARCHAR(64),
    created_at  TIMESTAMPTZ DEFAULT NOW()
);

```

## 2. API Key 管理

适用于三方系统调用（如服务端定时任务、外部集成等）。

### 2.1 数据库设计

```sql
CREATE TABLE sys_api_key (
    id          BIGSERIAL PRIMARY KEY,
    name        VARCHAR(64) NOT NULL,           -- 用途说明（如：ERP 集成）
    key_prefix  VARCHAR(8) NOT NULL,            -- 明文前缀（展示用，如 sk_rm_）
    key_hash    VARCHAR(64) UNIQUE NOT NULL,    -- SHA256 hash（不存明文）

    last_used_at TIMESTAMPTZ,
    status      SMALLINT DEFAULT 1,
    created_at  TIMESTAMPTZ DEFAULT NOW(),
    is_deleted  BOOLEAN DEFAULT FALSE
);
```

### 2.2 API Key 生成与验证

```python
# app/modules/auth/api_key.py
import secrets
import hashlib

class APIKeyService:

    @staticmethod
    def generate() -> tuple[str, str, str]:
        """
        生成 API Key
        Returns: (full_key, prefix, hash)
        full_key 格式：sk_rm_{随机32字节hex}
        """
        raw = secrets.token_hex(32)
        prefix = "sk_rm_"
        full_key = f"{prefix}{raw}"
        key_hash = hashlib.sha256(full_key.encode()).hexdigest()
        return full_key, prefix, key_hash

    async def verify(self, api_key: str, db) -> APIKeyRecord | None:
        """验证 API Key，返回对应记录"""
        key_hash = hashlib.sha256(api_key.encode()).hexdigest()
        record = await APIKeyRepository(db).get_by_hash(key_hash)
        if not record or not record.status:
            return None
        if record.expire_at and record.expire_at < datetime.utcnow():
            return None
        # 更新最后使用时间（非阻塞）
        asyncio.create_task(
            APIKeyRepository(db).update_last_used(record.id)
        )
        return record


# FastAPI 依赖：支持 Bearer Token 和 API Key 双认证
async def get_current_principal(
    authorization: str = Header(None),
    x_api_key: str = Header(None),
    db = Depends(get_db),
):
    if x_api_key:
        record = await APIKeyService().verify(x_api_key, db)
        if not record:
            raise HTTPException(status_code=401, detail="API Key 无效")
        user = await UserRepository(db).get_by_id(record.user_id)
        return user
    elif authorization and authorization.startswith("Bearer "):
        token = authorization[7:]
        return await get_current_user_from_token(token, db)
    raise HTTPException(status_code=401, detail="未提供认证凭据")
```

## 3. 登录保护

### 3.1 失败次数限制 + 账号锁定

```python
# app/modules/auth/service.py

LOGIN_FAIL_MAX = 5          # 最大失败次数
LOCK_DURATION_MINUTES = 30  # 锁定时长

async def login(self, username: str, password: str, db, redis) -> TokenPair:
    user = await UserRepository(db).get_by_username(username)

    if not user:
        raise AuthError("用户名或密码错误")

    # 检查是否锁定
    if user.locked_until and user.locked_until > datetime.utcnow():
        remaining = int((user.locked_until - datetime.utcnow()).total_seconds() / 60)
        raise AuthError(f"账号已锁定，请 {remaining} 分钟后再试")

    # 验证密码
    if not verify_password(password, user.password):
        # 记录失败
        fail_count = user.login_fail_count + 1
        update_data = {"login_fail_count": fail_count}
        if fail_count >= LOGIN_FAIL_MAX:
            update_data["locked_until"] = datetime.utcnow() + timedelta(minutes=LOCK_DURATION_MINUTES)
            # 写安全日志
            asyncio.create_task(
                SecurityLogService.record(
                    event_type="account_locked",
                    user_id=user.id,
                    username=username,
                    detail={"fail_count": fail_count},
                    risk_level="high",
                )
            )
        await UserRepository(db).update(user.id, update_data)
        raise AuthError("用户名或密码错误")

    # 登录成功：重置失败计数
    await UserRepository(db).update(user.id, {
        "login_fail_count": 0,
        "locked_until": None,
        "last_login": datetime.utcnow(),
    })

    return await self._issue_tokens(user, db)
```

### 3.2 登录图像验证码

所有登录请求强制验证图像验证码，防止暴力破解。

```python
# app/modules/auth/captcha.py

class CaptchaService:
    """验证码服务"""

    async def generate_captcha(self) -> dict:
        """生成验证码并返回图片和key"""
        captcha_code = self._generate_code()
        captcha_key = self._generate_key()

        await self.redis.setex(
            f"captcha:{captcha_key}",
            timedelta(minutes=5),
            captcha_code.lower(),
        )

        image_data = self._generate_image(captcha_code)

        return {
            "key": captcha_key,
            "image": image_data,
        }

    async def verify_captcha(self, captcha_key: str, captcha_code: str) -> bool:
        """校验验证码，校验成功后立即删除"""
        if not captcha_key or not captcha_code:
            return False

        stored_code = await self.redis.get(f"captcha:{captcha_key}")
        if not stored_code:
            return False

        await self.redis.delete(f"captcha:{captcha_key}")

        return stored_code == captcha_code.lower()
```

**特点**：
- 4位字母数字混合验证码
- 使用几何线条绘制字符，不依赖字体文件
- 随机干扰线和噪点增加识别难度
- 点击图片刷新验证码
- 验证码有效期5分钟
- 校验成功后立即删除，防止重复使用

**API 接口**：

| 方法 | 路径 | 说明 |
|------|------|------|
| GET | /api/v1/auth/captcha | 获取验证码（返回 key 和 base64 图片） |

**登录请求参数**：

| 参数 | 类型 | 必填 | 说明 |
|------|------|------|------|
| username | string | 是 | 用户名 |
| password | string | 是 | 密码 |
| captcha_key | string | 是 | 验证码 key |
| captcha_code | string | 是 | 用户输入的验证码 |

## 4. 文件上传安全

### 4.1 白名单策略

```python
# app/modules/file/validator.py
import magic  # python-magic

ALLOWED_EXTENSIONS = {
    "image": {".jpg", ".jpeg", ".png", ".gif", ".webp", ".svg"},
    "document": {".pdf", ".docx", ".xlsx", ".pptx", ".txt", ".md"},
    "video": {".mp4", ".mov", ".avi"},
    "audio": {".mp3", ".wav", ".m4a"},
}

ALLOWED_MIME_TYPES = {
    "image/jpeg", "image/png", "image/gif", "image/webp",
    "application/pdf",
    "application/vnd.openxmlformats-officedocument.wordprocessingml.document",
    "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
    "text/plain", "text/markdown",
    "video/mp4",
}

MAX_FILE_SIZE_MB = 100  # 最大 100MB

class FileValidator:

    @staticmethod
    async def validate(file_content: bytes, filename: str, category: str = None) -> None:
        """双重校验：文件扩展名 + 文件内容 Magic Bytes"""

        # 1. 大小检查
        size_mb = len(file_content) / (1024 * 1024)
        if size_mb > MAX_FILE_SIZE_MB:
            raise FileValidationError(f"文件大小超过限制（最大 {MAX_FILE_SIZE_MB}MB）")

        # 2. 扩展名检查（白名单）
        import os
        ext = os.path.splitext(filename)[1].lower()
        allowed = set()
        if category:
            allowed = ALLOWED_EXTENSIONS.get(category, set())
        else:
            for exts in ALLOWED_EXTENSIONS.values():
                allowed.update(exts)
        if ext not in allowed:
            raise FileValidationError(f"不允许的文件类型: {ext}")

        # 3. 文件内容 Magic Bytes 检测（防止伪造扩展名）
        detected_mime = magic.from_buffer(file_content[:2048], mime=True)
        if detected_mime not in ALLOWED_MIME_TYPES:
            raise FileValidationError(f"文件内容类型不匹配: {detected_mime}")
```

### 4.2 签名下载（私有文件访问控制）

```python
# app/modules/file/service.py

class FileService:

    async def generate_download_url(
        self,
        file_id: int,
        current_user,
        db,
        expires_seconds: int = 3600,
    ) -> str:
        """
        生成带签名的临时下载 URL（私有文件访问控制）
        需要校验当前用户是否有权限访问该文件
        """
        file_record = await FileRepository(db, current_user.tenant_id).get_by_id(file_id)
        if not file_record:
            raise PermissionError("文件不存在或无权访问")

        # 权限校验：文件归属 + 业务权限
        if not await self._can_access(current_user, file_record, db):
            raise PermissionError("无权访问该文件")

        # 根据存储类型生成签名 URL
        if settings.OSS_TYPE == "minio":
            return await self._minio_presigned_url(file_record.storage_path, expires_seconds)
        elif settings.OSS_TYPE == "aliyun":
            return await self._aliyun_presigned_url(file_record.storage_path, expires_seconds)

    async def _can_access(self, user, file_record, db) -> bool:
        """文件访问权限检查：创建者 or 租户内公开 or 有显式授权"""
        if file_record.created_by == user.id:
            return True
        if file_record.visibility == "tenant" and file_record.tenant_id == user.tenant_id:
            return True
        # 其他自定义业务规则...
        return False
```

## 5. CSRF 防护

> 适用于使用 Cookie 存储 Token 的场景（HTTP-only Cookie 存 Refresh Token）。

```python
# app/core/middleware/csrf.py
import secrets
from starlette.middleware.base import BaseHTTPMiddleware

CSRF_EXEMPT_PATHS = ["/api/v1/auth/login", "/api/v1/auth/refresh", "/api/v1/health"]

class CSRFMiddleware(BaseHTTPMiddleware):
    """
    Double Submit Cookie 方案：
    - 服务端登录时写入 csrf_token Cookie（非 HTTP-only，允许 JS 读取）
    - 前端每次请求在 Header X-CSRF-Token 中带上该值
    - 中间件验证 Cookie 值 == Header 值
    """

    async def dispatch(self, request: Request, call_next):
        # 只对状态变更请求（非 GET/HEAD/OPTIONS）校验
        if request.method in ("GET", "HEAD", "OPTIONS"):
            return await call_next(request)

        # 豁免路径
        if any(request.url.path.startswith(p) for p in CSRF_EXEMPT_PATHS):
            return await call_next(request)

        # 纯 Bearer Token 认证模式（Authorization Header）不需要 CSRF 保护
        if request.headers.get("Authorization", "").startswith("Bearer "):
            return await call_next(request)

        csrf_cookie = request.cookies.get("csrf_token")
        csrf_header = request.headers.get("X-CSRF-Token")

        if not csrf_cookie or csrf_cookie != csrf_header:
            return JSONResponse(
                status_code=403,
                content={"code": 403, "msg": "CSRF 验证失败"}
            )

        return await call_next(request)
```

## 6. 安全响应头

```python
# app/core/middleware/security_headers.py
from starlette.middleware.base import BaseHTTPMiddleware

class SecurityHeadersMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        response = await call_next(request)
        response.headers["X-Content-Type-Options"] = "nosniff"
        response.headers["X-Frame-Options"] = "DENY"
        response.headers["X-XSS-Protection"] = "1; mode=block"
        response.headers["Referrer-Policy"] = "strict-origin-when-cross-origin"
        response.headers["Permissions-Policy"] = "camera=(), microphone=(), geolocation=()"
        if not settings.DEBUG:
            response.headers["Strict-Transport-Security"] = "max-age=31536000; includeSubDomains"
        return response
```
