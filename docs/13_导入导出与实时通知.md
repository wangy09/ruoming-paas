## 一、数据导入导出

### 1.1 设计原则

| 场景 | 方案 |
|------|------|
| 小批量（< 1000 行）| 同步导出，直接返回文件流 |
| 大批量（> 1000 行）| 异步导出任务 + 进度推送 + 文件下载 |
| 数据导入 | 上传文件 → 异步校验 + 导入 → 进度推送 + 错误报告 |

```python
# app/modules/export/service.py
import asyncio
from openpyxl import Workbook

        await arq_queue.enqueue("process_export_task", task.id)

# app/tasks/export_task.py（arq worker 任务）
async def process_export_task(ctx, task_id: int):
    """
    异步导出任务（arq worker 执行）
    支持百万行：分批查询 + 流式写入
    """
    async with get_db() as db:
        task = await ExportTaskRepository(db).get_by_id(task_id)
        await ExportTaskRepository(db).update(task_id, {
            "status": "processing",
            "started_at": datetime.utcnow(),
        })

        try:
            # 1. 统计总行数
            total = await _count_data(task.task_type, task.params, db, task.tenant_id)
            await ExportTaskRepository(db).update(task_id, {"total_rows": total})

            # 2. 创建 Excel 工作簿
            wb = Workbook(write_only=True)  # write_only 模式：低内存流式写入
            ws = wb.create_sheet()

            # 写入表头
            headers = get_export_headers(task.task_type)
            ws.append(headers)

            # 3. 分批查询写入（每批 500 行，防止内存溢出）
            BATCH_SIZE = 500
            offset = 0
            exported = 0

            while True:
                rows = await _fetch_data_batch(
                    task.task_type, task.params, db, task.tenant_id,
                    offset=offset, limit=BATCH_SIZE
                )
                if not rows:
                    break

                for row in rows:
                    ws.append(row)
                    exported += 1

                offset += BATCH_SIZE

                # 更新进度（每批次更新）
                await ExportTaskRepository(db).update(task_id, {
                    "exported_rows": exported
                })
                # 通过 WebSocket 推送进度
                await _push_export_progress(task.user_id, task_id, exported, total)

            # 4. 保存到临时文件 → 上传 OSS
            import tempfile
            with tempfile.NamedTemporaryFile(suffix=".xlsx", delete=False) as f:
                wb.save(f.name)
                tmp_path = f.name

            oss_path = await upload_to_oss(tmp_path, f"exports/{task_id}.xlsx")
            os.unlink(tmp_path)  # 清理临时文件

            # 5. 更新任务完成
            await ExportTaskRepository(db).update(task_id, {
                "status": "done",
                "file_path": oss_path,
                "exported_rows": exported,
                "finished_at": datetime.utcnow(),
                "expire_at": datetime.utcnow() + timedelta(hours=24),
            })

            # 推送完成通知
            await _push_export_complete(task.user_id, task_id, oss_path)

        except Exception as e:
            await ExportTaskRepository(db).update(task_id, {
                "status": "failed",
                "error_msg": str(e),
                "finished_at": datetime.utcnow(),
            })
            raise
```

## 二、WebSocket 实时通知

### 2.1 通知类型

| 通知类型 | 触发场景 | 推送内容 |
|---------|---------|---------|
| `export_progress` | 导出任务进度更新 | 进度百分比 |
| `export_complete` | 导出任务完成 | 下载链接 |
| `import_complete` | 导入任务完成 | 成功/失败统计 |
| `ai_task_complete` | AI 工作流执行完成 | 结果摘要 |
| `system_notice` | 系统公告 | 通知内容 |
| `message` | 站内消息 | 消息内容 |

### 2.2 WebSocket 连接管理

```python
# app/core/websocket/manager.py
import asyncio
from fastapi import WebSocket

class ConnectionManager:
    """
    WebSocket 连接管理器
    按 user_id 维护多连接（同一用户多个浏览器 Tab）
    使用 Redis Pub/Sub 实现多进程广播
    """

    def __init__(self):
        # 本进程内的连接：{ user_id: [websocket, ...] }
        self._connections: dict[int, list[WebSocket]] = {}

    async def connect(self, websocket: WebSocket, user_id: int):
        await websocket.accept()
        if user_id not in self._connections:
            self._connections[user_id] = []
        self._connections[user_id].append(websocket)

    async def disconnect(self, websocket: WebSocket, user_id: int):
        if user_id in self._connections:
            self._connections[user_id] = [
                ws for ws in self._connections[user_id] if ws != websocket
            ]
            if not self._connections[user_id]:
                del self._connections[user_id]

    async def send_to_user(self, user_id: int, message: dict):
        """向本进程中指定用户的所有连接发送消息"""
        sockets = self._connections.get(user_id, [])
        dead_sockets = []
        for ws in sockets:
            try:
                await ws.send_json(message)
            except Exception:
                dead_sockets.append(ws)
        # 清理断连的 socket
        for ws in dead_sockets:
            await self.disconnect(ws, user_id)

    async def broadcast_to_tenant(self, tenant_id: int, message: dict):
        """向本进程中指定租户的所有用户广播（结合 Redis Pub/Sub 跨进程）"""
        for uid, sockets in list(self._connections.items()):
            # 这里需要知道每个 user_id 对应的 tenant_id（可从 Redis 缓存读取）
            user_tenant = await get_user_tenant_from_cache(uid)
            if user_tenant == tenant_id:
                await self.send_to_user(uid, message)


# 全局单例
ws_manager = ConnectionManager()
```

### 2.3 多进程广播（Redis Pub/Sub）

```python
# app/core/websocket/broadcaster.py
import json
import redis.asyncio as aioredis

CHANNEL_PREFIX = "ws:notify"

class WebSocketBroadcaster:
    """
    通过 Redis Pub/Sub 实现跨 Worker 进程的 WebSocket 推送
    发送方：publish 消息到 Redis
    接收方：各 Worker 订阅，收到消息后推给本进程内的 WebSocket 连接
    """

    @staticmethod
    async def publish_to_user(user_id: int, message: dict, redis):
        """向指定用户发布消息（跨进程）"""
        channel = f"{CHANNEL_PREFIX}:user:{user_id}"
        await redis.publish(channel, json.dumps(message))

    @staticmethod
    async def publish_to_tenant(tenant_id: int, message: dict, redis):
        """向指定租户广播（跨进程）"""
        channel = f"{CHANNEL_PREFIX}:tenant:{tenant_id}"
        await redis.publish(channel, json.dumps(message))

    @staticmethod
    async def start_subscriber(redis_url: str):
        """应用启动时开启订阅任务（每个 Worker 进程各一个）"""
        redis = aioredis.from_url(redis_url)
        pubsub = redis.pubsub()

        # 订阅所有用户通知 channel
        await pubsub.psubscribe(f"{CHANNEL_PREFIX}:user:*")
        await pubsub.psubscribe(f"{CHANNEL_PREFIX}:tenant:*")

        async for message in pubsub.listen():
            if message["type"] not in ("pmessage", "message"):
                continue
            channel = message.get("channel", b"").decode()
            data = json.loads(message["data"])

            if ":user:" in channel:
                user_id = int(channel.split(":user:")[1])
                await ws_manager.send_to_user(user_id, data)
            elif ":tenant:" in channel:
                tenant_id = int(channel.split(":tenant:")[1])
                await ws_manager.broadcast_to_tenant(tenant_id, data)
```

### 2.4 WebSocket 路由

```python
# app/modules/notification/ws_router.py
from fastapi import APIRouter, WebSocket, WebSocketDisconnect, Depends
from app.core.websocket.manager import ws_manager
from app.core.security import verify_token_from_ws

router = APIRouter()

@router.websocket("/ws/notify")
async def websocket_notify(websocket: WebSocket):
    """
    WebSocket 通知连接
    前端连接时通过 query 参数传 token：ws://host/ws/notify?token=xxx
    """
    token = websocket.query_params.get("token")
    try:
        payload = verify_token(token)
        user_id = int(payload["sub"])
    except Exception:
        await websocket.close(code=4001, reason="认证失败")
        return

    await ws_manager.connect(websocket, user_id)
    try:
        while True:
            # 保持连接，等待客户端 ping
            data = await websocket.receive_text()
            if data == "ping":
                await websocket.send_text("pong")
    except WebSocketDisconnect:
        await ws_manager.disconnect(websocket, user_id)
```

### 2.5 前端 WebSocket 封装

```typescript
// composables/useWebSocket.ts
import { ref, onMounted, onUnmounted } from 'vue'
import { getToken } from '@/utils/auth'

export function useWebSocket() {
  const socket = ref<WebSocket | null>(null)
  const isConnected = ref(false)

  const connect = () => {
    const token = getToken()
    const wsUrl = `${import.meta.env.VITE_WS_URL}/ws/notify?token=${token}`
    socket.value = new WebSocket(wsUrl)

    socket.value.onopen = () => {
      isConnected.value = true
      // 心跳保活（每 30s 发送 ping）
      heartbeat = setInterval(() => socket.value?.send('ping'), 30000)
    }

    socket.value.onmessage = (event) => {
      const message = JSON.parse(event.data)
      handleMessage(message)
    }

    socket.value.onclose = () => {
      isConnected.value = false
      clearInterval(heartbeat)
      // 断线重连（3s 后重试，最多 5 次）
      setTimeout(connect, 3000)
    }
  }

  const handleMessage = (message: { type: string; data: any }) => {
    switch (message.type) {
      case 'export_progress':
        // 更新导出进度条
        useExportStore().updateProgress(message.data)
        break
      case 'export_complete':
        ElNotification.success({ title: '导出完成', message: '点击下载文件' })
        break
      case 'ai_task_complete':
        ElNotification.success({ title: 'AI 任务完成', message: message.data.summary })
        break
      case 'system_notice':
        ElNotification.warning({ title: '系统通知', message: message.data.content })
        break
      case 'message':
        // 站内消息红点 +1
        useNotificationStore().addUnread(message.data)
        break
    }
  }

  let heartbeat: ReturnType<typeof setInterval>

  onMounted(connect)
  onUnmounted(() => {
    clearInterval(heartbeat)
    socket.value?.close()
  })

  return { isConnected }
}
```

### 3.1 支持的协议

| 协议/平台 | 适用场景 |
|---------|---------|
| OAuth2 + OIDC | 标准 SSO，对接任意 OIDC Provider（Keycloak/Azure AD/Google） |

| 钉钉 | 国内企业客户 |

### 3.2 OIDC 接入流程

```
用户点击 "企业 SSO 登录"
        │
        ▼
后端生成 state（防 CSRF）+ 返回 OIDC Provider 授权 URL
        │
        ▼（浏览器跳转到企业 IdP 登录页）
用户在 IdP 完成登录
        │
        ▼（IdP 回调带 code + state）
后端验证 state → 用 code 换 id_token + access_token
        │
        ▼
解析 id_token，获取用户信息（email/name/sub）
        │
    ┌───┴───┐
    ▼       ▼
用户已存在  首次登录
直接登录   自动创建账号 → 绑定企业用户
    │       │
    └───┬───┘
        ▼
签发本系统 JWT（Access + Refresh Token）
        │
        ▼
前端存储 Token，跳转首页
```

### 3.4 OIDC 认证路由

```python
# app/modules/auth/sso_router.py
from fastapi import APIRouter, Request
from authlib.integrations.httpx_client import AsyncOAuth2Client

router = APIRouter(prefix="/auth/sso")

@router.get("/providers")
async def list_sso_providers(request: Request, db=Depends(get_db)):
    """获取当前租户可用的 SSO Provider 列表（用于登录页展示）"""
    tenant_id = request.state.tenant_id
    return await SSOProviderRepository(db).get_by_tenant(tenant_id)

@router.get("/{provider_id}/authorize")
async def sso_authorize(provider_id: int, db=Depends(get_db), redis=Depends(get_redis)):
    """生成 SSO 授权 URL，返回给前端跳转"""
    provider = await SSOProviderRepository(db).get_by_id(provider_id)
    state = secrets.token_urlsafe(32)
    await redis.setex(f"sso:state:{state}", 600, str(provider_id))  # 10分钟有效

    client = AsyncOAuth2Client(
        client_id=provider.client_id,
        client_secret=decrypt(provider.client_secret),
        scope=provider.scopes,
    )
    url, _ = client.create_authorization_url(
        provider.auth_url,
        state=state,
        redirect_uri=f"{settings.BASE_URL}/auth/sso/{provider_id}/callback"
    )
    return {"authorize_url": url}

@router.get("/{provider_id}/callback")
async def sso_callback(
    provider_id: int,
    code: str,
    state: str,
    db=Depends(get_db),
    redis=Depends(get_redis),
):
    """SSO 回调处理"""
    # 验证 state 防 CSRF
    stored_provider_id = await redis.get(f"sso:state:{state}")
    if not stored_provider_id or int(stored_provider_id) != provider_id:
        raise AuthError("无效的 state 参数")
    await redis.delete(f"sso:state:{state}")

    provider = await SSOProviderRepository(db).get_by_id(provider_id)
    client = AsyncOAuth2Client(
        client_id=provider.client_id,
        client_secret=decrypt(provider.client_secret),
    )
    # 换取 token
    tokens = await client.fetch_token(
        provider.token_url,
        code=code,
        redirect_uri=f"{settings.BASE_URL}/auth/sso/{provider_id}/callback"
    )
    # 获取用户信息
    userinfo = await client.get(provider.userinfo_url)
    userinfo_data = userinfo.json()

    # 查找或创建用户
    user = await SSOService(db).find_or_create_user(
        provider_id=provider_id,
        external_id=userinfo_data.get("sub"),
        email=userinfo_data.get("email"),
        name=userinfo_data.get("name"),
        tenant_id=provider.tenant_id,
        auto_register=provider.auto_register,
    )

    # 签发系统 Token
    return await AuthService(db).issue_tokens(user)
```
