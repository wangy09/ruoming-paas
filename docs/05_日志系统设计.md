# 05 日志系统设计

## 1. 日志分类体系

所有日志按类型分类存储，方便查询和审计。

| 日志类型 | 说明 | 存储位置 | 保留期 |
|---------|------|---------|--------|
| 访问日志 | 每次 HTTP 请求记录 | 文件 + 日志系统 | 30天 |
| 操作日志 | 用户增删改操作（含变更前后快照） | 数据库 + 文件 | 1年 |
| 登录日志 | 登录/登出/失败记录 | 数据库 + 文件 | 180天 |
| 系统日志 | 程序运行状态、错误 | 文件 + 日志系统 | 30天 |
| 任务日志 | 异步任务执行 | 数据库 + 文件 | 90天 |
| AI 调用日志 | LLM 调用记录 | 数据库 | 90天 |
| 安全日志 | 权限拒绝、异常访问 | 数据库 + 文件 | 1年 |
| 审计日志 | 敏感数据访问、配置变更 | 数据库 | 3年 |

## 2. 日志库选型（二选一，不并存）

> **v2 决策**：原设计同时引入 `structlog` 和 `Loguru`，两套日志库并存配置复杂且可能冲突。本版本统一方案如下：
>
> - **`structlog`** 作为主日志库：负责结构化 JSON 渲染、上下文变量绑定（RequestID/tenant_id）、与标准 `logging` 集成
> - **`Loguru`** 作为文件输出 sink：只负责文件轮转、压缩、保留策略，通过 `logging` 标准库桥接接收 structlog 的输出
> - 业务代码统一使用 `get_logger()` 获取 structlog logger，不直接调用 Loguru

```
structlog （结构化处理）→ stdlib logging （桥接）→ Loguru sink（文件轮转/压缩）
                                                 → 控制台输出（开发环境）
                                                 → 日志采集器（Loki/ELK，生产环境）
```

## 3. 结构化日志实现

### 3.1 日志配置（统一入口）

```python
# app/core/logging.py
import logging
import sys
import structlog
from loguru import logger as loguru_logger
from app.config import settings


def setup_logging():
    """
    初始化日志系统：
    - structlog 负责结构化处理和上下文绑定
    - Loguru 负责文件轮转/压缩输出
    - 通过 stdlib logging 桥接
    """
    # ── 1. Loguru 文件 sink 配置 ──────────────────────────────
    _setup_loguru_sinks()

    # ── 2. 将标准 logging 重定向到 Loguru ──────────────────────
    # 移除默认 handler
    logging.basicConfig(handlers=[], level=logging.DEBUG if settings.DEBUG else logging.INFO)

    class InterceptHandler(logging.Handler):
        """把 stdlib logging 消息转发给 Loguru"""
        def emit(self, record: logging.LogRecord) -> None:
            try:
                level = loguru_logger.level(record.levelname).name
            except ValueError:
                level = record.levelno
            frame, depth = sys._getframe(6), 6
            while frame and frame.f_code.co_filename == logging.__file__:
                frame = frame.f_back
                depth += 1
            loguru_logger.opt(depth=depth, exception=record.exc_info).log(
                level, record.getMessage()
            )

    logging.root.handlers = [InterceptHandler()]

    # ── 3. structlog 配置 ──────────────────────────────────────
    shared_processors = [
        structlog.contextvars.merge_contextvars,
        structlog.stdlib.add_log_level,
        structlog.stdlib.add_logger_name,
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.StackInfoRenderer(),
    ]

    if settings.ENVIRONMENT == "production":
        processors = shared_processors + [
            structlog.processors.format_exc_info,
            structlog.processors.JSONRenderer(),
        ]
    else:
        processors = shared_processors + [
            structlog.dev.ConsoleRenderer(colors=True),
        ]

    structlog.configure(
        processors=processors,
        context_class=dict,
        logger_factory=structlog.stdlib.LoggerFactory(),
        wrapper_class=structlog.make_filtering_bound_logger(
            logging.DEBUG if settings.DEBUG else logging.INFO
        ),
        cache_logger_on_first_use=True,
    )


def get_logger(name: str = None):
    """统一 logger 获取入口，业务代码只调用此函数"""
    return structlog.get_logger(name)


def _setup_loguru_sinks():
    """配置 Loguru 文件分类输出"""
    # 移除 Loguru 默认 stderr 输出
    loguru_logger.remove()

    base_dir = settings.LOG_DIR  # 默认 /var/log/ruoming-saas

    loguru_logger.add(
        f"{base_dir}/access/access_{{time:YYYY-MM-DD}}.log",
        filter=lambda r: r["extra"].get("log_type") == "access",
        format="{message}",
        rotation="00:00", retention="30 days", compression="gz", serialize=True
    )
    loguru_logger.add(
        f"{base_dir}/system/system_{{time:YYYY-MM-DD}}.log",
        filter=lambda r: r["extra"].get("log_type") != "access",
        level="INFO",
        rotation="100 MB", retention="30 days", compression="gz", serialize=True
    )
    loguru_logger.add(
        f"{base_dir}/error/error_{{time:YYYY-MM-DD}}.log",
        level="ERROR",
        rotation="00:00", retention="90 days", compression="gz",
        serialize=True, backtrace=True, diagnose=True
    )
    loguru_logger.add(
        f"{base_dir}/ai/ai_call_{{time:YYYY-MM-DD}}.log",
        filter=lambda r: r["extra"].get("log_type") == "ai_call",
        rotation="00:00", retention="90 days", compression="gz", serialize=True
    )
    loguru_logger.add(
        f"{base_dir}/security/security_{{time:YYYY-MM-DD}}.log",
        filter=lambda r: r["extra"].get("log_type") == "security",
        rotation="00:00", retention="365 days", compression="gz", serialize=True
    )
```

### 3.2 RequestID 中间件

```python
# app/core/middleware/request_id.py
import uuid
import structlog
from starlette.middleware.base import BaseHTTPMiddleware
from starlette.requests import Request

class RequestIDMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        request_id = request.headers.get("X-Request-ID", str(uuid.uuid4()))
        
        structlog.contextvars.clear_contextvars()
        structlog.contextvars.bind_contextvars(
            request_id=request_id,
            path=request.url.path,
            method=request.method,
        )
        
        response = await call_next(request)
        response.headers["X-Request-ID"] = request_id
        return response
```

### 3.3 请求访问日志中间件

```python
# app/core/middleware/logging.py
import time
import structlog
from starlette.middleware.base import BaseHTTPMiddleware

logger = structlog.get_logger("access")

class AccessLogMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        start_time = time.perf_counter()
        
        try:
            response = await call_next(request)
            duration_ms = int((time.perf_counter() - start_time) * 1000)
            
            logger.info(
                "http_request",
                status_code=response.status_code,
                duration_ms=duration_ms,
                client_ip=request.client.host,
                user_agent=request.headers.get("User-Agent", ""),
                log_type="access",
            )
            return response
        except Exception as exc:
            duration_ms = int((time.perf_counter() - start_time) * 1000)
            logger.error("http_request_error", error=str(exc), duration_ms=duration_ms)
            raise
```

## 4. 操作日志（数据库存储）

### 4.1 数据库表（含变更前后快照）

> **v2 新增**：`sys_operation_log` 增加 `before_data` 字段，记录数据变更前的值。审计场景（如"谁将用户A的角色从普通用户改为管理员"）需要前后对比才有意义。

```sql
CREATE TABLE sys_operation_log (
    id              BIGSERIAL PRIMARY KEY,
    user_id         BIGINT,
    username        VARCHAR(64),
    module          VARCHAR(64),                -- 操作模块（如：用户管理）
    action          VARCHAR(64),                -- 操作动作（如：修改角色）
    method          VARCHAR(10),                -- HTTP 方法
    url             VARCHAR(256),               -- 请求 URL
    ip              VARCHAR(64),                -- 客户端 IP
    location        VARCHAR(128),               -- IP 归属地
    browser         VARCHAR(128),
    os              VARCHAR(64),
    request_param   TEXT,                       -- 请求参数（脱敏后）
    before_data     JSONB,                      -- 变更前数据快照（UPDATE/DELETE 场景）
    after_data      JSONB,                      -- 变更后数据快照
    response_result TEXT,                       -- 响应结果摘要
    status          SMALLINT DEFAULT 1,         -- 1成功 0失败
    error_msg       TEXT,
    cost_time       INT,                        -- 耗时（ms）
    request_id      VARCHAR(64),
    tenant_id       BIGINT,
    created_at      TIMESTAMPTZ DEFAULT NOW()
);

-- 登录日志
CREATE TABLE sys_login_log (
    id          BIGSERIAL PRIMARY KEY,
    user_id     BIGINT,
    username    VARCHAR(64),
    login_type  VARCHAR(20),                    -- password/sso/oauth
    ip          VARCHAR(64),
    location    VARCHAR(128),
    browser     VARCHAR(128),
    os          VARCHAR(64),
    status      SMALLINT DEFAULT 1,             -- 1成功 0失败
    msg         VARCHAR(256),                   -- 失败原因
    login_time  TIMESTAMPTZ DEFAULT NOW(),
    tenant_id   BIGINT
);

-- 安全日志
CREATE TABLE sys_security_log (
    id          BIGSERIAL PRIMARY KEY,
    event_type  VARCHAR(64),                    -- permission_denied/brute_force/unusual_access/account_locked
    user_id     BIGINT,
    username    VARCHAR(64),
    ip          VARCHAR(64),
    url         VARCHAR(256),
    detail      JSONB,
    risk_level  VARCHAR(20) DEFAULT 'low',      -- low/medium/high/critical
    request_id  VARCHAR(64),
    created_at  TIMESTAMPTZ DEFAULT NOW()
);
```

### 4.2 操作日志装饰器（非阻塞 + 变更前后快照）

> **v2 修复**：
> 1. 原设计 `await OperationLogService.record(...)` 仍会等待写日志协程完成再返回响应，实质上是阻塞的。修复为 `asyncio.create_task(...)` 真正非阻塞。
> 2. 新增 `capture_before` 参数，UPDATE/DELETE 前自动取快照，写入 `before_data`。

```python
# app/core/decorators/operation_log.py
import asyncio
import functools
import time
from app.core.logging import get_logger
from app.modules.log.service import OperationLogService

logger = get_logger("operation")

def operation_log(
    module: str,
    action: str,
    sensitive_fields: list[str] = None,
    capture_before: bool = False,           # 是否在执行前捕获数据快照（用于 UPDATE/DELETE）
    before_loader: callable = None,         # 捕获前置数据的异步函数，签名：async (args, kwargs) -> dict
):
    """
    操作日志装饰器
    用法：
        @operation_log(module="用户管理", action="修改角色",
                       capture_before=True, before_loader=load_user_before)
        async def update_user_role(...)
    """
    def decorator(func):
        @functools.wraps(func)
        async def wrapper(*args, **kwargs):
            request = _extract_request(args, kwargs)
            current_user = _extract_user(args, kwargs)
            start = time.perf_counter()

            # 变更前快照
            before_data = None
            if capture_before and before_loader:
                try:
                    before_data = await before_loader(args, kwargs)
                except Exception:
                    pass  # 快照失败不影响主流程

            try:
                result = await func(*args, **kwargs)
                cost_ms = int((time.perf_counter() - start) * 1000)

                # ✅ 真正非阻塞：create_task 丢入事件循环，不等待完成即返回
                asyncio.create_task(
                    OperationLogService.record(
                        user_id=current_user.id if current_user else None,
                        username=current_user.username if current_user else None,
                        module=module,
                        action=action,
                        method=request.method if request else None,
                        url=str(request.url) if request else None,
                        ip=request.client.host if request else None,
                        request_param=_safe_params(request, sensitive_fields),
                        before_data=before_data,
                        after_data=_extract_after_data(result),
                        status=1,
                        cost_time=cost_ms,
                    )
                )

                logger.info("operation", module=module, action=action,
                           user=current_user.username if current_user else None,
                           cost_ms=cost_ms)
                return result

            except Exception as e:
                cost_ms = int((time.perf_counter() - start) * 1000)
                # 失败日志同样非阻塞
                asyncio.create_task(
                    OperationLogService.record(
                        user_id=current_user.id if current_user else None,
                        username=current_user.username if current_user else None,
                        module=module,
                        action=action,
                        before_data=before_data,
                        status=0,
                        error_msg=str(e),
                        cost_time=cost_ms,
                    )
                )
                logger.error("operation_failed", module=module, action=action, error=str(e))
                raise

        return wrapper
    return decorator
```

# app/modules/user/router.py

async def _load_user_before(args, kwargs):
    """在修改前加载用户当前数据作为快照"""

    db = kwargs.get("db")
    user = await UserService(db).get_by_id(user_id)
    if user:
        return {"roles": [r.code for r in user.roles], "status": user.status}
    return None


@router.post("/", dependencies=[Depends(require_permissions("system:user:create"))])

## 6. 链路追踪（OpenTelemetry）

```python
# app/core/tracing.py
from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor
from opentelemetry.exporter.otlp.proto.grpc.trace_exporter import OTLPSpanExporter
from opentelemetry.instrumentation.fastapi import FastAPIInstrumentor
from opentelemetry.instrumentation.sqlalchemy import SQLAlchemyInstrumentor
from opentelemetry.instrumentation.redis import RedisInstrumentor

def setup_tracing(app):
    if not settings.OTEL_ENABLED:
        return
    
    provider = TracerProvider()
    exporter = OTLPSpanExporter(endpoint=settings.OTEL_ENDPOINT)
    provider.add_span_processor(BatchSpanProcessor(exporter))
    trace.set_tracer_provider(provider)
    
    FastAPIInstrumentor.instrument_app(app)
    SQLAlchemyInstrumentor().instrument()
    RedisInstrumentor().instrument()
```
