## 2. Axios 请求封装

```typescript
// src/utils/request.ts
import axios, { AxiosInstance, AxiosRequestConfig } from 'axios'
import { ElMessage } from 'element-plus'
import { useAuthStore } from '@/store/auth'
import router from '@/router'

const request: AxiosInstance = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 30000,
  headers: { 'Content-Type': 'application/json' },
})

// 请求拦截器：注入 Token + Tenant
request.interceptors.request.use(config => {
  const authStore = useAuthStore()
  if (authStore.token) {
    config.headers['Authorization'] = `Bearer ${authStore.token}`
  }
  if (authStore.tenantCode) {
    config.headers['X-Tenant-Code'] = authStore.tenantCode
  }
  return config
})

// 响应拦截器：统一错误处理 + Token 刷新
let isRefreshing = false
let pendingRequests: Array<() => void> = []

request.interceptors.response.use(
  response => {
    const res = response.data
    if (res.code !== 200) {
      ElMessage.error(res.message || '请求失败')
      return Promise.reject(new Error(res.message))
    }
    return res  // 返回 ApiResponse
  },
  async error => {
    const { response, config } = error
    if (response?.status === 401) {
      // Token 过期，尝试刷新
      if (!isRefreshing) {
        isRefreshing = true
        try {
          const authStore = useAuthStore()
          await authStore.refreshToken()
          pendingRequests.forEach(cb => cb())
          pendingRequests = []
          return request(config)  // 重试原请求
        } catch {
          await useAuthStore().logout()
          router.push('/login')
        } finally {
          isRefreshing = false
        }
      }
      // 等待刷新完成后重试
      return new Promise(resolve => {
        pendingRequests.push(() => resolve(request(config)))
      })
    }
    
    if (response?.status === 403) {
      ElMessage.error('无操作权限')
    } else if (response?.status >= 500) {
      ElMessage.error('服务器异常，请稍后重试')
    }
    
    return Promise.reject(error)
  }
)

export default request
```

## 3. ProTable 封装组件（支持列权限）

```typescript
// src/components/ProTable/types.ts
export interface ColumnConfig {
  prop: string                    // 字段名
  label: string                   // 列头
  width?: number | string
  minWidth?: number | string
  fixed?: 'left' | 'right'
  sortable?: boolean
  formatter?: (row: any) => string
  render?: (row: any) => VNode    // 自定义渲染
  permissionCode?: string         // 该列的权限 code（可见性控制）
  editable?: boolean              // 是否可行内编辑
  hide?: boolean                  // 默认隐藏（用户可自行显示）
  searchable?: boolean            // 是否支持搜索
}
```

```vue
<!-- src/components/ProTable/index.vue -->
<template>
  <div class="pro-table">
    <!-- 工具栏 -->
    <div class="table-toolbar">
      <slot name="actions" />
      <div class="table-toolbar-right">
        <el-button :icon="Refresh" circle @click="refresh" />
        <ColumnSetting :columns="allColumns" v-model="visibleProps" />
      </div>
    </div>

    <!-- 表格主体 -->
    <el-table
      v-loading="loading"
      :data="tableData"
      v-bind="$attrs"
      @selection-change="handleSelectionChange"
    >
      <el-table-column v-if="selection" type="selection" width="55" />
      
      <template v-for="col in permittedColumns" :key="col.prop">
        <el-table-column
          v-if="visibleProps.includes(col.prop)"
          :prop="col.prop"
          :label="col.label"
          :width="col.width"
          :fixed="col.fixed"
          :sortable="col.sortable"
        >
          <template #default="{ row }">
            <slot :name="col.prop" :row="row">
              {{ col.formatter ? col.formatter(row) : row[col.prop] }}
            </slot>
          </template>
        </el-table-column>
      </template>
    </el-table>

    <!-- 分页 -->
    <el-pagination
      v-if="pagination"
      class="table-pagination"
      v-model:current-page="currentPage"
      v-model:page-size="pageSize"
      :total="total"
      :page-sizes="[10, 20, 50, 100]"
      layout="total, sizes, prev, pager, next, jumper"
      @change="handlePageChange"
    />
  </div>
</template>

<script setup lang="ts">
import { computed, ref, onMounted } from 'vue'
import { usePermissionStore } from '@/store/permission'
import { Refresh } from '@element-plus/icons-vue'
import type { ColumnConfig } from './types'

const props = withDefaults(defineProps<{
  resourceKey?: string       // 列权限资源标识
  columns: ColumnConfig[]
  fetchData: (params: any) => Promise<{ items: any[], total: number }>
  selection?: boolean
  pagination?: boolean
  immediate?: boolean        // 是否立即加载
}>(), {
  pagination: true,
  immediate: true,
})

const permStore = usePermissionStore()

// 过滤无权限的列
const permittedColumns = computed(() => {
  if (!props.resourceKey) return props.columns
  const allowed = permStore.getVisibleColumns(props.resourceKey)
  if (!allowed.length) return props.columns
  return props.columns.filter(col =>
    !col.permissionCode || allowed.includes(col.prop)
  )
})

// 用户自定义显示列（持久化到 localStorage）
const storageKey = `table_cols_${props.resourceKey}`
const visibleProps = ref<string[]>(
  JSON.parse(localStorage.getItem(storageKey) || 'null')
  ?? permittedColumns.value.filter(c => !c.hide).map(c => c.prop)
)
</script>
```

### 4.2 动态路由构建

```typescript
// src/router/guard.ts
const modules = import.meta.glob('@/views/**/*.vue')

function buildRoutes(menus: MenuItem[]): RouteRecordRaw[] {
  return menus.map(menu => {
    const route: RouteRecordRaw = {
      path: menu.path,
      name: menu.component,
      component: menu.component
        ? modules[`/src/views/${menu.component}.vue`]
        : () => import('@/layouts/DefaultLayout.vue'),
      meta: {
        title: menu.title,
        icon: menu.icon,
        keepAlive: menu.keepAlive ?? false,
        permissionCode: menu.permissionCode,
        menuId: menu.id,
      },
    }
    if (menu.children?.length) {
      route.children = buildRoutes(menu.children)
    }
    return route
  })
}
```
