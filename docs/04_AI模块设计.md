# 04 AI 模块设计

## 1. 设计目标

- **统一 LLM 网关**：追鹿大模型为主，兼容 OpenAI 协议，可切换任意模型
- **可插拔 Agent**：每个 Agent 独立定义，工具/提示词/模型可配置
- **Multi-Agent 编排**：支持串行/并行/条件分支的**有向图（含受控回退循环）**流程
- **知识库 RAG**：支持多格式文档解析、**混合检索（向量 + BM25 + Rerank）**
- **工具链扩展**：内置常用工具，支持自定义工具注册
- **全链路可观测**：每次推理记录输入/输出/Token/耗时/Cost

### 3.2 追鹿大模型接入

```python
# app/ai/llm/providers/zhuilu.py
import httpx
from app.ai.llm.providers.base import BaseLLMProvider

class ZhuiluProvider(BaseLLMProvider):
    """
    追鹿大模型 Provider
    使用 OpenAI 兼容协议（如果追鹿支持），否则适配其原生 API
    """

    def __init__(self, api_key: str, base_url: str, **kwargs):
        self.api_key = api_key
        self.base_url = base_url.rstrip("/")
        self.client = httpx.AsyncClient(
            base_url=self.base_url,
            headers={"Authorization": f"Bearer {api_key}"},
            timeout=120.0
        )

    async def chat(self, messages, model, temperature=0.7,
                   max_tokens=None, stream=False, **kwargs):
        payload = {
            "model": model,
            "messages": messages,
            "temperature": temperature,
            "stream": stream,
        }
        if max_tokens:
            payload["max_tokens"] = max_tokens

        if stream:
            return self._stream_chat(payload)
        
        resp = await self.client.post("/v1/chat/completions", json=payload)
        resp.raise_for_status()
        data = resp.json()
        return data["choices"][0]["message"]["content"]

    async def _stream_chat(self, payload):
        async with self.client.stream("POST", "/v1/chat/completions", json=payload) as resp:
            async for line in resp.aiter_lines():
                if line.startswith("data: ") and line != "data: [DONE]":
                    import json
                    chunk = json.loads(line[6:])
                    delta = chunk["choices"][0].get("delta", {})
                    if content := delta.get("content"):
                        yield content

    async def embed(self, texts: list[str], model: str) -> list[list[float]]:
        resp = await self.client.post("/v1/embeddings", json={
            "model": model,
            "input": texts
        })
        resp.raise_for_status()
        data = resp.json()
        return [item["embedding"] for item in data["data"]]
```

### 3.4 LLM 服务层（含多进程缓存失效）

> **v2 修复**：原设计用类变量 `_providers` 在进程内存缓存 Provider 实例。多 Worker 部署时，动态修改模型配置后，其他进程的缓存不会自动失效。修复方案：配置变更时推送 Redis Pub/Sub 事件，各 Worker 订阅后主动清除本地缓存。

```python
# app/ai/llm/service.py
import asyncio
import redis.asyncio as aioredis
from app.ai.llm.providers.base import BaseLLMProvider
from app.ai.llm.providers.zhuilu import ZhuiluProvider
from app.ai.llm.providers.openai import OpenAIProvider

PROVIDER_CACHE_CHANNEL = "llm:config:invalidate"

class LLMService:
    # 进程内缓存（仅当前 Worker 有效）
    _providers: dict[int, BaseLLMProvider] = {}

    @classmethod
    async def invalidate_provider(cls, model_config_id: int):
        """从本进程缓存中移除指定配置的 Provider"""
        cls._providers.pop(model_config_id, None)

    @classmethod
    async def start_invalidation_listener(cls, redis_url: str):
        """
        在应用启动时开启后台任务，监听配置变更消息，
        及时清除本进程缓存，保证多 Worker 配置一致性
        """
        redis = aioredis.from_url(redis_url)

        async def _listen():
            pubsub = redis.pubsub()
            await pubsub.subscribe(PROVIDER_CACHE_CHANNEL)
            async for message in pubsub.listen():
                if message["type"] == "message":
                    config_id = int(message["data"])
                    await cls.invalidate_provider(config_id)

        asyncio.create_task(_listen())

    @classmethod
    async def get_provider(cls, model_config_id: int, db) -> BaseLLMProvider:
        """根据配置 ID 获取（并缓存）Provider 实例"""
        if model_config_id not in cls._providers:
            config = await ModelConfigRepository(db).get_by_id(model_config_id)
            cls._providers[model_config_id] = cls._build_provider(config)
        return cls._providers[model_config_id]

    @staticmethod
    def _build_provider(config) -> BaseLLMProvider:
        match config.provider:
            case "zhuilu":
                return ZhuiluProvider(
                    api_key=decrypt(config.api_key),
                    base_url=config.base_url
                )
            case "openai":
                return OpenAIProvider(api_key=decrypt(config.api_key))
            case _:
                raise ValueError(f"未知 Provider: {config.provider}")

    async def chat(self, messages: list[dict], model_config_id: int,
                   db, **kwargs) -> str:
        """统一对话入口，自动记录调用日志"""
        provider = await self.get_provider(model_config_id, db)
        config = await ModelConfigRepository(db).get_by_id(model_config_id)

        start = time.time()
        try:
            result = await provider.chat(messages, config.model_id, **kwargs)
            await self._save_log(model_config_id, messages, result,
                                 time.time() - start, "success", db)
            return result
        except Exception as e:
            await self._save_log(model_config_id, messages, str(e),
                                 time.time() - start, "error", db)
            raise

    @classmethod
    async def notify_config_changed(cls, model_config_id: int, redis_url: str):
        """模型配置变更时调用，广播给所有 Worker"""
        redis = aioredis.from_url(redis_url)
        await redis.publish(PROVIDER_CACHE_CHANNEL, str(model_config_id))
```

### 4.2 Agent 执行器（含超时控制）

```python
# app/ai/agent/executor.py
import asyncio
from langchain.agents import AgentExecutor, create_openai_tools_agent
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder

class AgentRunner:
    """单 Agent 执行器（封装 LangChain，不向外暴露 LangChain 类型）"""

    def __init__(self, agent_config: AgentConfig, tool_registry: ToolRegistry):
        self.config = agent_config
        self.tool_registry = tool_registry

    async def run(
        self,
        user_message: str,
        session_id: str,
        conversation_history: list[dict],
        stream: bool = False,
        timeout_sec: int | None = None,
    ) -> dict:
        # 超时时间：优先使用调用方传入，其次使用 Agent 配置，最后默认 60s
        timeout = timeout_sec or self.config.node_timeout_sec or 60

        try:
            result = await asyncio.wait_for(
                self._do_run(user_message, session_id, conversation_history, stream),
                timeout=timeout,
            )
            return result
        except asyncio.TimeoutError:
            raise AgentTimeoutError(
                f"Agent [{self.config.name}] 执行超时（{timeout}s）"
            )

    async def _do_run(self, user_message, session_id, conversation_history, stream):
        tools = self.tool_registry.get_tools(self.config.tool_ids)

        prompt = ChatPromptTemplate.from_messages([
            ("system", self.config.system_prompt),
            MessagesPlaceholder("chat_history"),
            ("human", "{input}"),
            MessagesPlaceholder("agent_scratchpad"),
        ])

        llm = await build_langchain_llm(self.config.model_config_id)

        agent = create_openai_tools_agent(llm, tools, prompt)
        executor = AgentExecutor(
            agent=agent,
            tools=tools,
            verbose=True,
            max_iterations=self.config.max_rounds,
            return_intermediate_steps=True,
        )

        if stream:
            return executor.astream_events(
                {"input": user_message, "chat_history": conversation_history},
                version="v2"
            )
        else:
            result = await executor.ainvoke(
                {"input": user_message, "chat_history": conversation_history}
            )
            return result
```

## 5. Multi-Agent 编排（`ai/multi_agent`）

### 5.1 工作流图类型说明

> **v2 澄清**：工作流图类型为**有向图（Directed Graph）**，而非严格 DAG。
>
> - **默认路径**是无环 DAG（拓扑排序执行）
> - **条件回退**（如 `condition_check → agent_write` 的 false 边）构成受控循环
> - 为防止无限循环，每个循环路径设有**最大迭代次数**（`max_loop_count`，默认 3）
> - 执行引擎在运行时追踪每个节点的执行次数，超出限制立即报错终止

### 5.2 工作流定义（含版本化）

### 5.4 编排器实现（含超时 + 并发安全 + 循环控制）

```python
# app/ai/multi_agent/orchestrator.py
import asyncio
from asyncio import Lock
from typing import Any
from copy import deepcopy

class WorkflowOrchestrator:
    """Multi-Agent 工作流编排器（有向图，支持受控循环）"""

    def __init__(self, workflow: Workflow, graph_json: dict, db, tool_registry: ToolRegistry):
        self.workflow = workflow
        self.graph = graph_json
        self.db = db
        self.tool_registry = tool_registry
        # 线程/协程安全的上下文字典（并行节点写入需加锁）
        self._context: dict[str, Any] = {}
        self._context_lock = Lock()
        # 追踪节点执行次数（防止无限循环）
        self._node_exec_count: dict[str, int] = {}

    async def _set_context(self, key: str, value: Any):
        async with self._context_lock:
            self._context[key] = value

    async def _get_context(self) -> dict[str, Any]:
        async with self._context_lock:
            return deepcopy(self._context)

    async def run(self, input_data: dict, run_id: int) -> dict:
        graph = self.graph
        node_map = {n["id"]: n for n in graph["nodes"]}
        await self._set_context("start", input_data)

        # 使用图遍历（BFS + 条件边选择）代替简单拓扑排序，支持循环
        current_node_id = "start"
        while current_node_id != "end":
            node = node_map[current_node_id]
            next_node_id = await self._run_node_and_get_next(node, run_id, graph)
            if next_node_id is None:
                break
            current_node_id = next_node_id

        ctx = await self._get_context()
        return ctx.get("end", {})

    async def _run_node_and_get_next(self, node: dict, run_id: int, graph: dict) -> str | None:
        node_id = node["id"]

        # 循环次数检查
        self._node_exec_count[node_id] = self._node_exec_count.get(node_id, 0) + 1
        max_loop = node.get("config", {}).get("max_loop_count", 3)
        if self._node_exec_count[node_id] > max_loop:
            raise WorkflowLoopError(
                f"节点 [{node_id}] 执行次数超过最大限制 {max_loop}，终止工作流"
            )

        node_type = node["type"]
        condition_result = None

        match node_type:
            case "start":
                pass  # 已在 run() 初始化
            case "agent":
                timeout = node.get("config", {}).get("timeout_sec", 60)
                result = await asyncio.wait_for(
                    self._run_agent_node(node),
                    timeout=timeout,
                )
                await self._set_context(node_id, result)
            case "tool":
                result = await self._run_tool_node(node)
                await self._set_context(node_id, result)
            case "condition":
                ctx = await self._get_context()
                condition_result = self._evaluate_condition(node, ctx)
                await self._set_context(node_id, {"result": condition_result})
            case "parallel":
                # 并行执行：所有子节点并发运行，结果分别存入 context
                await self._run_parallel_nodes(node, run_id)

        # 选择下一个节点
        return self._pick_next_node(node_id, graph, condition_result)

    def _pick_next_node(self, current_id: str, graph: dict, condition_result=None) -> str | None:
        edges = [e for e in graph["edges"] if e["from"] == current_id]
        if not edges:
            return None
        for edge in edges:
            edge_cond = edge.get("condition")
            if edge_cond is None:
                return edge["to"]
            # 条件边：匹配条件结果
            if str(condition_result).lower() == edge_cond.lower():
                return edge["to"]
        return None

    async def _run_parallel_nodes(self, parallel_node: dict, run_id: int):
        """并行执行多个子节点，结果安全写入 context"""
        sub_node_ids = parallel_node.get("config", {}).get("nodes", [])
        graph = self.graph
        node_map = {n["id"]: n for n in graph["nodes"]}

        tasks = [
            self._run_agent_node(node_map[nid])
            for nid in sub_node_ids
            if nid in node_map
        ]
        results = await asyncio.gather(*tasks, return_exceptions=True)

        for nid, result in zip(sub_node_ids, results):
            if isinstance(result, Exception):
                await self._set_context(nid, {"error": str(result)})
            else:
                await self._set_context(nid, result)

    async def _run_agent_node(self, node: dict) -> dict:
        config = node["config"]
        agent_config = await load_agent_config(config["agent_id"], self.db)
        ctx = await self._get_context()
        input_text = self._render_template(config["input_template"], ctx)
        runner = AgentRunner(agent_config, self.tool_registry)
        result = await runner.run(input_text, session_id=str(node["id"]))
        return {"output": result["output"], "steps": result.get("intermediate_steps")}

    def _render_template(self, template: str, context: dict) -> str:
        import re
        def replace(m):
            keys = m.group(1).strip().split(".")
            val = context
            for k in keys:
                val = val.get(k, "") if isinstance(val, dict) else ""
            return str(val)
        return re.sub(r"\{\{(.+?)\}\}", replace, template)

    def _evaluate_condition(self, node: dict, context: dict) -> bool:
        """安全地对条件表达式求值（限制执行环境）"""
        condition = node.get("config", {}).get("condition", "false")
        # 模板变量替换后求值（使用受限 eval，仅允许比较运算）
        rendered = self._render_template(condition, context)
        try:
            # 只允许比较运算，不允许函数调用
            return bool(eval(rendered, {"__builtins__": {}}, {}))
        except Exception:
            return False
```

### 6.2 文档增量更新策略

> **v2 新增**：文档修改后，旧 chunk 的向量需要清理，否则检索结果会包含过时内容。通过 `content_hash` 判断是否真正变化。

```python
# app/ai/knowledge/service.py

async def update_document(self, doc_id: int, new_file_content: bytes):
    """
    文档增量更新：
    1. 计算新内容哈希，若与旧哈希相同则跳过
    2. 删除旧 chunk 的向量（向量库 + 数据库）
    3. 重新分块、向量化、写入
    """
    import hashlib
    new_hash = hashlib.sha256(new_file_content).hexdigest()

    doc = await self.doc_repo.get_by_id(doc_id)
    if doc.content_hash == new_hash:
        return {"message": "内容未变化，跳过更新"}

    # 1. 删除旧 chunks 在向量库中的向量
    old_chunks = await self.chunk_repo.get_by_doc(doc_id)
    if old_chunks:
        old_vector_ids = [c.vector_id for c in old_chunks if c.vector_id]
        await self.vector_store.delete(
            collection=f"kb_{doc.kb_id}",
            ids=old_vector_ids
        )

    # 2. 删除数据库中旧 chunks
    await self.chunk_repo.delete_by_doc(doc_id)

    # 3. 重新索引
    pipeline = await self._get_pipeline(doc.kb_id)
    await pipeline.index_document(doc, new_file_content)

    # 4. 更新 hash
    await self.doc_repo.update(doc_id, {"content_hash": new_hash})
    return {"message": "文档更新成功"}
```

### 6.3 混合检索 Pipeline（Hybrid Search）

> **v2 新增**：纯向量检索在词汇精确匹配场景（如专有名词、代码）效果不佳。混合检索结合向量相似度 + BM25 全文检索，再通过 Rerank 模型重排序，显著提升 RAG 召回质量。

```python
# app/ai/knowledge/pipeline.py
class RAGPipeline:
    """RAG 处理管道：解析 → 分块 → 向量化 → 存储 → 混合检索"""

    def __init__(self, kb_config: KnowledgeBase, vector_store, embed_service, rerank_service=None):
        self.kb = kb_config
        self.vector_store = vector_store
        self.embed_service = embed_service
        self.rerank_service = rerank_service   # 可选 Rerank 模型

    # ── 索引阶段 ──────────────────────────────

    async def index_document(self, doc: KnowledgeDoc, file_content: bytes):
        """文档 → 向量索引（Celery 异步任务中执行）"""
        text = await self._parse_document(doc.source_type, file_content)
        chunks = self._split_text(text, self.kb.chunk_size, self.kb.chunk_overlap)

        embeddings = await self.embed_service.embed_batch(
            [c.content for c in chunks],
            self.kb.embed_model_id
        )

        vector_ids = await self.vector_store.upsert(
            collection=f"kb_{self.kb.id}",
            ids=[c.id for c in chunks],
            embeddings=embeddings,
            metadata=[{"doc_id": doc.id, "content": c.content} for c in chunks]
        )
        return vector_ids

    # ── 检索阶段（混合检索）──────────────────────────────

    async def retrieve(self, query: str, top_k: int = None) -> list[RetrievedChunk]:
        """
        混合检索：向量检索 + BM25 全文检索，可选 Rerank 重排序
        retrieval_type: vector / bm25 / hybrid
        """
        top_k = top_k or self.kb.top_k

        if self.kb.retrieval_type == "vector":
            return await self._vector_search(query, top_k)

        elif self.kb.retrieval_type == "bm25":
            return await self._bm25_search(query, top_k)

        else:  # hybrid（默认）
            # 1. 两路并行召回（各取 top_k * 2，确保候选集足够大）
            vector_results, bm25_results = await asyncio.gather(
                self._vector_search(query, top_k * 2),
                self._bm25_search(query, top_k * 2),
            )

            # 2. RRF 融合（Reciprocal Rank Fusion）去重合并
            merged = self._rrf_merge(vector_results, bm25_results)

            # 3. Rerank 重排序（若配置了 Rerank 模型）
            if self.rerank_service and self.kb.rerank_model_id:
                merged = await self.rerank_service.rerank(
                    query=query,
                    docs=merged,
                    model_id=self.kb.rerank_model_id,
                    top_k=top_k,
                )
            else:
                merged = merged[:top_k]

            return merged

    async def _vector_search(self, query: str, top_k: int) -> list[RetrievedChunk]:
        """向量相似度检索"""
        query_embedding = await self.embed_service.embed(query, self.kb.embed_model_id)
        return await self.vector_store.search(
            collection=f"kb_{self.kb.id}",
            vector=query_embedding,
            top_k=top_k,
            score_threshold=self.kb.score_threshold,
        )

    async def _bm25_search(self, query: str, top_k: int) -> list[RetrievedChunk]:
        """BM25 全文检索（基于 Elasticsearch 或 PostgreSQL FTS）"""
        # 使用 PostgreSQL 全文搜索（无需额外 ES 组件）
        from sqlalchemy import text
        sql = text("""
            SELECT id, content, ts_rank(to_tsvector('chinese', content), plainto_tsquery(:query)) AS score
            FROM ai_knowledge_chunk
            WHERE kb_id = :kb_id
              AND to_tsvector('chinese', content) @@ plainto_tsquery(:query)
            ORDER BY score DESC
            LIMIT :top_k
        """)
        # ... 执行并返回 RetrievedChunk 列表

    def _rrf_merge(
        self,
        list1: list[RetrievedChunk],
        list2: list[RetrievedChunk],
        k: int = 60,
    ) -> list[RetrievedChunk]:
        """
        Reciprocal Rank Fusion 融合两路检索结果
        score = 1/(k + rank)，k 默认 60（经验值）
        """
        scores: dict[str, float] = {}
        chunks: dict[str, RetrievedChunk] = {}

        for rank, chunk in enumerate(list1):
            scores[chunk.id] = scores.get(chunk.id, 0) + 1 / (k + rank + 1)
            chunks[chunk.id] = chunk

        for rank, chunk in enumerate(list2):
            scores[chunk.id] = scores.get(chunk.id, 0) + 1 / (k + rank + 1)
            chunks[chunk.id] = chunk

        sorted_ids = sorted(scores, key=lambda x: scores[x], reverse=True)
        return [chunks[cid] for cid in sorted_ids]

    async def retrieve_and_format(self, query: str) -> str:
        """检索并格式化为 LLM 上下文"""
        chunks = await self.retrieve(query)
        context = "\n\n---\n\n".join([c.content for c in chunks])
        return f"以下是相关参考资料：\n\n{context}"

    def _split_text(self, text: str, chunk_size: int, overlap: int) -> list[Chunk]:
        from langchain.text_splitter import RecursiveCharacterTextSplitter
        splitter = RecursiveCharacterTextSplitter(
            chunk_size=chunk_size,
            chunk_overlap=overlap,
            separators=["\n\n", "\n", "。", "！", "？", " ", ""]
        )
        texts = splitter.split_text(text)
        return [Chunk(content=t) for t in texts]
```

## 11. 场景路由（Scene Router）

> 同一个 Agent 或对话框，根据用户意图/场景动态切换 Prompt、temperature、模型、RAG策略。采用**混合模式**：前端显式传入优先，规则引擎兜底，模型分类按需。

### 11.1 场景优先级链

```
scene_type 优先级（从高到低）：
①  前端显式传入 scene_type（最准确，推荐业务方使用）
②  Agent 绑定的默认场景（ai_agent.default_scene）
③  规则引擎推断（关键词 / 长度 / 特征，覆盖 ~80% 请求）
④  轻量模型分类（规则兜不住时，可选，有成本）
⑤  系统兜底（professional）
```

### 11.2 内置场景类型

| scene_type | 含义 | 推荐 temperature | 推荐模型规格 |
|------------|------|-----------------|------------|
| `chitchat` | 日常闲聊、问候 | 0.9 | 轻量模型 |
| `professional` | 专业问答、分析 | 0.2 | 主力模型（默认） |
| `task` | 任务执行（写代码/写报告） | 0.3 | 主力模型 |
| `rag` | 基于知识库的文档问答 | 0.1 | 主力模型 |
| `creative` | 创意写作、头脑风暴 | 1.0 | 主力模型 |

### 11.4 规则引擎（轻量推断，覆盖 ~80% 场景）

```python
# app/ai/scene/router.py

import re
from dataclasses import dataclass

PROFESSIONAL_KEYWORDS = re.compile(
    r"分析|报告|方案|代码|实现|优化|比较|评估|原理|解释|总结|翻译|写作", re.I
)
TASK_KEYWORDS = re.compile(
    r"帮我写|生成|创建|实现|设计|制作|出一份|做一个", re.I
)
RAG_KEYWORDS = re.compile(
    r"根据文档|文件中|资料里|知识库|上传的|附件", re.I
)

@dataclass
class SceneRouteResult:
    scene_type: str
    source: str       # explicit / agent_default / rule / model / fallback


class SceneRouter:
    """
    场景路由器：混合模式
    优先级：前端显式 > Agent默认 > 规则引擎 > 模型分类 > 兜底
    """

    def __init__(self, classifier_llm=None):
        self._classifier = classifier_llm  # 可选的轻量分类模型

    async def resolve(
        self,
        message: str,
        *,
        explicit_scene: str | None = None,       # 前端传入
        agent_default_scene: str | None = None,  # Agent 绑定的默认场景
        has_attachment: bool = False,
        use_model_classifier: bool = False,       # 是否启用模型分类（成本开关）
    ) -> SceneRouteResult:

        # ① 前端显式传入
        if explicit_scene:
            return SceneRouteResult(scene_type=explicit_scene, source="explicit")

        # ② Agent 绑定默认场景
        if agent_default_scene:
            return SceneRouteResult(scene_type=agent_default_scene, source="agent_default")

        # ③ 规则引擎（无 LLM 调用，零成本）
        rule_result = self._rule_infer(message, has_attachment)
        if rule_result:
            return SceneRouteResult(scene_type=rule_result, source="rule")

        # ④ 模型分类（可选，需显式开启）
        if use_model_classifier and self._classifier:
            model_result = await self._model_classify(message)
            if model_result:
                return SceneRouteResult(scene_type=model_result, source="model")

        # ⑤ 兜底
        return SceneRouteResult(scene_type="professional", source="fallback")

    def _rule_infer(self, message: str, has_attachment: bool) -> str | None:
        """规则引擎：顺序匹配，命中即返回"""
        if has_attachment:
            return "rag"
        if RAG_KEYWORDS.search(message):
            return "rag"
        if TASK_KEYWORDS.search(message):
            return "task"
        if PROFESSIONAL_KEYWORDS.search(message):
            return "professional"
        # 短消息（≤15字且无专业词汇）→ 闲聊
        if len(message.strip()) <= 15:
            return "chitchat"
        return None  # 规则无法判断，交给下一层

    async def _model_classify(self, message: str) -> str | None:
        """轻量模型意图分类（仅规则兜不住时调用）"""
        prompt = f"""将以下用户消息分类到一个场景类型，只返回类型名，不要解释。
场景类型：chitchat / professional / task / rag / creative

用户消息：{message[:200]}

场景类型："""
        try:
            result = await self._classifier.chat(
                messages=[{"role": "user", "content": prompt}],
                model="fast",     # 使用轻量/快速模型
                temperature=0,
                max_tokens=10,
            )
            scene = result.strip().lower()
            return scene if scene in ("chitchat", "professional", "task", "rag", "creative") else None
        except Exception:
            return None
```

### 11.5 接入 Agent 执行器

```python
# app/ai/agent/executor.py（调用链改造）

class AgentRunner:

    async def run(
        self,
        user_message: str,
        session_id: str,
        conversation_history: list[dict],
        stream: bool = False,
        timeout_sec: int | None = None,
        scene_type: str | None = None,      # ← 新增：前端传入的场景
        has_attachment: bool = False,       # ← 新增：是否带附件
    ) -> dict:

        # 1. 解析场景
        router = SceneRouter()
        scene = await router.resolve(
            message=user_message,
            explicit_scene=scene_type,
            agent_default_scene=self.config.default_scene,
            has_attachment=has_attachment,
        )

        # 2. 加载场景配置（DB 查询，有 Redis 缓存）
        scene_cfg = await SceneConfigService().get(
            agent_id=self.config.id,
            scene_type=scene.scene_type,
        )

        # 3. 用场景配置覆盖 Agent 默认参数
        effective_model_id  = scene_cfg.model_config_id or self.config.model_config_id
        effective_prompt    = scene_cfg.system_prompt   or self.config.system_prompt
        effective_temp      = scene_cfg.temperature     if scene_cfg.temperature is not None \
                              else self.config.temperature
        effective_max_tok   = scene_cfg.max_tokens      or self.config.max_tokens

        # 4. 若场景开启 RAG，先检索上下文
        rag_context = ""
        if scene_cfg.enable_rag and self.config.kb_ids:
            rag_context = await self._retrieve_rag(
                user_message,
                kb_ids=self.config.kb_ids,
                top_k=scene_cfg.rag_top_k,
                score_threshold=scene_cfg.rag_score_threshold,
            )
            if rag_context:
                effective_prompt += f"\n\n{rag_context}"

        # 5. 执行推理（沿用原有超时控制）
        timeout = timeout_sec or self.config.node_timeout_sec or 60
        result = await asyncio.wait_for(
            self._do_run(
                user_message, session_id, conversation_history, stream,
                model_id=effective_model_id,
                system_prompt=effective_prompt,
                temperature=effective_temp,
                max_tokens=effective_max_tok,
            ),
            timeout=timeout,
        )
        result["scene"] = scene.scene_type    # 结果中附带场景信息，便于前端展示/调试
        result["scene_source"] = scene.source
        return result
```

## 12. Prompt 模板版本化（`ai/prompt`）

> `ai.prompt` 模块负责管理可复用的 Prompt 模板，支持版本快照、变量插值、A/B 测试。Agent 和场景配置通过引用 prompt_id（而非直接存 system_prompt 文本）实现 Prompt 与配置解耦。

### 12.1 数据模型

```sql
-- Prompt 模板主表
CREATE TABLE ai_prompt_template (
    id          BIGSERIAL PRIMARY KEY,

    description TEXT,

);

-- Agent 引用 Prompt 版本（解耦 Agent 与 Prompt 内容）
ALTER TABLE ai_agent
    ADD COLUMN prompt_template_id BIGINT REFERENCES ai_prompt_template(id),
    ADD COLUMN prompt_version INT;    -- NULL=始终使用最新已发布版本，指定=锁定版本

-- 场景配置也可引用 Prompt
ALTER TABLE ai_scene_config
    ADD COLUMN prompt_template_id BIGINT REFERENCES ai_prompt_template(id),
    ADD COLUMN prompt_version INT;
```

### 12.2 版本管理策略

| 操作 | 行为 |
|------|------|
| 创建模板 | `version=1`，状态 `is_published=false`（草稿） |
| 修改内容 | 新增 `version+1` 行，旧版本保留不修改 |
| 发布版本 | `is_published=true`，此后可被 Agent 引用 |
| Agent 引用 | `prompt_version=NULL` → 始终跟随最新已发布版本（推荐）<br>`prompt_version=N` → 锁定指定版本（稳定性优先） |
| 回滚 | 将 current_version 改回旧版本号，不删数据 |

### 12.3 Prompt 渲染服务

```python
# app/ai/prompt/service.py

class PromptService:

    async def render(
        self,
        template_id: int,
        variables: dict,
        version: int | None = None,   # None=最新已发布版本
    ) -> str:
        """加载指定版本的 Prompt 模板并插值变量"""
        tmpl = await self._get_version(template_id, version)
        return self._interpolate(tmpl.content, variables)

    async def _get_version(self, template_id: int, version: int | None):
        """从缓存或DB加载版本内容"""
        cache_key = f"prompt:{template_id}:{version or 'latest'}"
        cached = await self.redis.get(cache_key)
        if cached:
            return json.loads(cached)

        if version is None:
            # 查最新已发布版本
            row = await self.db.execute(
                select(PromptVersion)
                .where(PromptVersion.template_id == template_id)
                .where(PromptVersion.is_published == True)
                .order_by(PromptVersion.version.desc())
                .limit(1)
            )
        else:
            row = await self.db.get(PromptVersion, {"template_id": template_id, "version": version})

        await self.redis.setex(cache_key, 300, json.dumps(row))  # 缓存 5 分钟
        return row

    def _interpolate(self, content: str, variables: dict) -> str:
        """{{variable}} 变量替换"""
        import re
        def replace(m):
            key = m.group(1).strip()
            return str(variables.get(key, f"{{{{ {key} }}}}"))  # 未找到则保留原占位符
        return re.sub(r"\{\{(.+?)\}\}", replace, content)
```
